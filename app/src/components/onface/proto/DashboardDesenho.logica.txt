
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.chartRef = React.createRef();
    this.paletteRef = React.createRef();
    this.state = {
      view: 'overview',
      range: props.intervaloInicial || '30d',
      metric: 'gross',
      showPrev: props.mostrarComparacao !== false,
      loading: false, hover: null, chartW: 900, tween: 0,
      query: '', status: 'todos', sortKey: 'date', sortDir: 'desc',
      limit: props.linhasPorPagina || 12,
      selectedId: null, confirming: false,
      paletteOpen: false, paletteQuery: '', paletteIdx: 0,
      chargeOpen: false, formAmount: '', formName: '', formMethod: 'Pix',
      formError: false, submitting: false,
      toast: null,
      payments: this.buildPayments()
    };
  }

  componentDidMount() {
    this.measure();
    this._ro = new ResizeObserver(() => this.measure());
    if (this.chartRef.current) this._ro.observe(this.chartRef.current);
    this._onKey = (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        this.setState(s => ({ paletteOpen: !s.paletteOpen, paletteQuery: '', paletteIdx: 0 }),
          () => { if (this.state.paletteOpen && this.paletteRef.current) this.paletteRef.current.focus(); });
      } else if (e.key === 'Escape') {
        this.setState({ paletteOpen: false, chargeOpen: false, selectedId: null, confirming: false });
      }
    };
    document.addEventListener('keydown', this._onKey);
    this.tween();
  }
  componentWillUnmount() {
    if (this._ro) this._ro.disconnect();
    document.removeEventListener('keydown', this._onKey);
    cancelAnimationFrame(this._raf);
    clearTimeout(this._t1); clearTimeout(this._t2); clearTimeout(this._t3);
  }
  measure() {
    const el = this.chartRef.current;
    if (el && el.clientWidth && Math.abs(el.clientWidth - this.state.chartW) > 2) this.setState({ chartW: el.clientWidth });
  }
  tween() {
    cancelAnimationFrame(this._raf);
    const t0 = performance.now();
    const step = (t) => {
      const k = Math.min(1, (t - t0) / 720);
      this.setState({ tween: 1 - Math.pow(1 - k, 3) });
      if (k < 1) this._raf = requestAnimationFrame(step);
    };
    this._raf = requestAnimationFrame(step);
  }
  flash(text, tone) {
    clearTimeout(this._t3);
    this.setState({ toast: { text, tone: tone || 'ok' } });
    this._t3 = setTimeout(() => this.setState({ toast: null }), 2800);
  }

  rng(seed) {
    let t = seed >>> 0;
    return () => {
      t = (t + 0x6D2B79F5) >>> 0;
      let x = Math.imul(t ^ (t >>> 15), 1 | t);
      x = (x + Math.imul(x ^ (x >>> 7), 61 | x)) ^ x;
      return ((x ^ (x >>> 14)) >>> 0) / 4294967296;
    };
  }

  brl(v) { return 'R$ ' + Number(v).toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); }
  brlShort(v) {
    if (v >= 1e6) return 'R$ ' + (v / 1e6).toLocaleString('pt-BR', { maximumFractionDigits: 2 }) + ' mi';
    if (v >= 1e3) return 'R$ ' + (v / 1e3).toLocaleString('pt-BR', { maximumFractionDigits: 1 }) + ' mil';
    return 'R$ ' + v.toLocaleString('pt-BR', { maximumFractionDigits: 0 });
  }
  int(v) { return Math.round(v).toLocaleString('pt-BR'); }
  pct(v) { return v.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + '%'; }

  // ---------- data ----------
  buildPayments() {
    const names = ['Ana Beatriz Moraes', 'Carlos Eduardo Lima', 'Fernanda Prado', 'Rodrigo Tavares', 'Juliana Campos',
      'MarceloBastos', 'Patrícia Rezende', 'Thiago Nogueira', 'Larissa Andrade', 'Vinícius Salgado',
      'Camila Duarte', 'Bruno Cerqueira', 'Renata Vasques', 'Eduardo Pontes', 'Mariana Castelo',
      'Felipe Arruda', 'Gabriela Menezes', 'Leandro Figueiró', 'Sofia Bittencourt', 'Otávio Peixoto',
      'Helena Quadros', 'Daniel Vilhena', 'Isabela Muniz', 'Gustavo Rocha', 'Natália Braga', 'André Sampaio'];
    const methods = ['Pix', 'Cartão de crédito', 'Cartão de crédito', 'Boleto', 'Cartão de débito', 'Cartão de crédito'];
    const statuses = ['Aprovado', 'Aprovado', 'Aprovado', 'Aprovado', 'Recusado', 'Reembolsado', 'Aprovado', 'Em análise'];
    const rnd = this.rng(20260922);
    const alpha = '23456789abcdefghjkmnpqrstuvwxyz';
    const base = new Date(2026, 8, 22, 14, 38);
    return names.map((n, i) => {
      const amount = Math.round((40 + Math.pow(rnd(), 2.1) * 4200) * 100) / 100;
      const status = statuses[Math.floor(rnd() * statuses.length)];
      const method = methods[Math.floor(rnd() * methods.length)];
      const d = new Date(base.getTime() - (i * 7.4 + rnd() * 6) * 3600 * 1000);
      let id = 'pay_';
      for (let k = 0; k < 14; k++) id += alpha[Math.floor(rnd() * alpha.length)];
      const first = n.split(' ')[0].toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
      const last = n.split(' ').slice(-1)[0].toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
      return {
        id, customer: n, email: first + '.' + last + '@' + (i % 3 === 0 ? 'gmail.com' : i % 3 === 1 ? 'outlook.com' : 'empresa.com.br'),
        amount, status, method, date: d, risk: status === 'Em análise' ? 'Elevado · 71' : status === 'Recusado' ? 'Médio · 48' : 'Normal · ' + Math.round(4 + rnd() * 20)
      };
    });
  }

  rangeConf() {
    return { '7d': { n: 7, step: 'day' }, '30d': { n: 30, step: 'day' }, '90d': { n: 13, step: 'week' }, '12m': { n: 12, step: 'month' } }[this.state.range];
  }
  series() {
    const conf = this.rangeConf();
    const rnd = this.rng({ '7d': 11, '30d': 22, '90d': 33, '12m': 44 }[this.state.range]);
    const mult = { day: 1, week: 6.6, month: 29 }[conf.step];
    const months = ['jan', 'fev', 'mar', 'abr', 'mai', 'jun', 'jul', 'ago', 'set', 'out', 'nov', 'dez'];
    const base = new Date(2026, 8, 22);
    const out = [];
    for (let i = 0; i < conf.n; i++) {
      const p = i / (conf.n - 1);
      const wave = Math.sin(i * 0.9) * 0.1 + Math.sin(i * 0.31) * 0.07;
      const gross = (18400 + p * 11200 + wave * 9000 + rnd() * 4200) * mult;
      const count = Math.round(gross / (168 + rnd() * 40));
      let label;
      if (conf.step === 'month') { const d = new Date(base.getFullYear(), base.getMonth() - (conf.n - 1 - i), 1); label = months[d.getMonth()] + ' ' + String(d.getFullYear()).slice(2); }
      else if (conf.step === 'week') { const d = new Date(base.getTime() - (conf.n - 1 - i) * 7 * 864e5); label = 'sem ' + d.getDate() + '/' + (d.getMonth() + 1); }
      else { const d = new Date(base.getTime() - (conf.n - 1 - i) * 864e5); label = d.getDate() + ' ' + months[d.getMonth()]; }
      out.push({
        label, gross, net: gross * 0.9578 - count * 0.39, count,
        decline: 4.1 + Math.sin(i * 0.7) * 0.8 + rnd() * 0.7,
        prevGross: gross * (0.79 + rnd() * 0.14)
      });
    }
    return out;
  }
  metricConf() {
    return {
      gross: { title: 'Volume bruto', key: 'gross', fmt: (v) => this.brl(v), axis: (v) => this.brlShort(v), agg: 'sum' },
      net: { title: 'Receita líquida', key: 'net', fmt: (v) => this.brl(v), axis: (v) => this.brlShort(v), agg: 'sum' },
      count: { title: 'Pagamentos aprovados', key: 'count', fmt: (v) => this.int(v), axis: (v) => this.int(v), agg: 'sum' },
      decline: { title: 'Taxa de recusa', key: 'decline', fmt: (v) => this.pct(v), axis: (v) => v.toFixed(1) + '%', agg: 'avg' }
    }[this.state.metric];
  }
  agg(arr, mode) { const s = arr.reduce((a, b) => a + b, 0); return mode === 'avg' ? s / arr.length : s; }
  path(pts, close, H) {
    if (!pts.length) return '';
    let d = 'M' + pts[0][0].toFixed(1) + ' ' + pts[0][1].toFixed(1);
    for (let i = 1; i < pts.length; i++) {
      const a = pts[i - 1], b = pts[i], cx = (a[0] + b[0]) / 2;
      d += ' C' + cx.toFixed(1) + ' ' + a[1].toFixed(1) + ' ' + cx.toFixed(1) + ' ' + b[1].toFixed(1) + ' ' + b[0].toFixed(1) + ' ' + b[1].toFixed(1);
    }
    if (close) d += ' L' + pts[pts.length - 1][0].toFixed(1) + ' ' + H + ' L' + pts[0][0].toFixed(1) + ' ' + H + ' Z';
    return d;
  }
  pill(status) {
    const map = {
      'Aprovado': ['#E6F4F1', '#0A6558'], 'Recusado': ['#FFF1F3', '#B02A41'],
      'Reembolsado': ['#F2F4F7', '#475467'], 'Em análise': ['#FDF3E4', '#9A5B0C']
    };
    const c = map[status] || map['Reembolsado'];
    return 'display:inline-flex;align-items:center;background:' + c[0] + ';color:' + c[1] + ';font-size:11.5px;font-weight:600;padding:3px 9px;border-radius:6px;white-space:nowrap';
  }
  initials(n) { const p = n.split(' '); return (p[0][0] + (p[1] ? p[1][0] : '')).toUpperCase(); }
  avatar(n) {
    const tints = [['#E6F4F1', '#0A6558'], ['#EEEBFA', '#5847B8'], ['#FDF3E4', '#8F5A12'], ['#EAF0F8', '#3A5A85'], ['#FDEEF1', '#A03149']];
    const c = tints[n.charCodeAt(0) % tints.length];
    return 'flex:0 0 auto;width:28px;height:28px;border-radius:50%;background:' + c[0] + ';color:' + c[1] + ';display:grid;place-items:center;font-size:10.5px;font-weight:600';
  }
  fmtDate(d) {
    const m = ['jan', 'fev', 'mar', 'abr', 'mai', 'jun', 'jul', 'ago', 'set', 'out', 'nov', 'dez'];
    return d.getDate() + ' ' + m[d.getMonth()] + ', ' + String(d.getHours()).padStart(2, '0') + ':' + String(d.getMinutes()).padStart(2, '0');
  }
  fullDate(d) {
    const m = ['janeiro', 'fevereiro', 'março', 'abril', 'maio', 'junho', 'julho', 'agosto', 'setembro', 'outubro', 'novembro', 'dezembro'];
    return d.getDate() + ' de ' + m[d.getMonth()] + ' de ' + d.getFullYear() + ' às ' + String(d.getHours()).padStart(2, '0') + ':' + String(d.getMinutes()).padStart(2, '0');
  }

  // ---------- actions ----------
  go(view) { this.setState({ view, query: '', status: 'todos', limit: this.props.linhasPorPagina || 12, paletteOpen: false }); }
  setRange(range) {
    this.setState({ range, loading: true, hover: null });
    clearTimeout(this._t1);
    this._t1 = setTimeout(() => { this.setState({ loading: false }); this.tween(); }, 420);
  }
  filtered() {
    const q = this.state.query.trim().toLowerCase();
    let rows = this.state.payments.filter(p => {
      if (this.state.status !== 'todos' && p.status !== this.state.status) return false;
      if (!q) return true;
      return (p.customer + ' ' + p.email + ' ' + p.id).toLowerCase().includes(q);
    });
    const dir = this.state.sortDir === 'asc' ? 1 : -1;
    const k = this.state.sortKey;
    rows.sort((a, b) => {
      if (k === 'amount') return (a.amount - b.amount) * dir;
      if (k === 'date') return (a.date - b.date) * dir;
      return String(a[k === 'customer' ? 'customer' : 'status']).localeCompare(String(b[k === 'customer' ? 'customer' : 'status']), 'pt-BR') * dir;
    });
    return rows;
  }
  sort(key) {
    this.setState(s => ({ sortKey: key, sortDir: s.sortKey === key ? (s.sortDir === 'asc' ? 'desc' : 'asc') : (key === 'customer' ? 'asc' : 'desc') }));
  }
  arrow(key) { return this.state.sortKey === key ? (this.state.sortDir === 'asc' ? ' ↑' : ' ↓') : ''; }
  exportCsv() {
    const rows = this.filtered();
    const head = 'id,cliente,email,valor,status,metodo,data\n';
    const body = rows.map(p => [p.id, '"' + p.customer + '"', p.email, p.amount.toFixed(2).replace('.', ','), p.status, '"' + p.method + '"', p.date.toISOString()].join(',')).join('\n');
    const url = URL.createObjectURL(new Blob([head + body], { type: 'text/csv;charset=utf-8' }));
    const a = document.createElement('a');
    a.href = url; a.download = 'onface-pagamentos.csv'; a.click();
    setTimeout(() => URL.revokeObjectURL(url), 1500);
    this.flash(rows.length + ' pagamentos exportados');
  }
  submitCharge() {
    const raw = this.state.formAmount.replace(/\./g, '').replace(',', '.');
    const val = parseFloat(raw);
    if (!val || !this.state.formName.trim()) { this.setState({ formError: true }); return; }
    this.setState({ formError: false, submitting: true });
    clearTimeout(this._t2);
    this._t2 = setTimeout(() => {
      const alpha = '23456789abcdefghjkmnpqrstuvwxyz';
      let id = 'pay_'; for (let k = 0; k < 14; k++) id += alpha[Math.floor(Math.random() * alpha.length)];
      const name = this.state.formName.trim();
      const first = name.split(' ')[0].toLowerCase();
      const p = { id, customer: name, email: first + '@cobranca.onface.app', amount: val, status: 'Em análise', method: this.state.formMethod, date: new Date(), risk: 'Aguardando · —' };
      this.setState(s => ({ payments: [p, ...s.payments], submitting: false, chargeOpen: false, formAmount: '', formName: '' }));
      this.flash('Cobrança de ' + this.brl(val) + ' criada', 'warn');
    }, 780);
  }
  refundConfirm() {
    const id = this.state.selectedId;
    this.setState(s => ({ payments: s.payments.map(p => p.id === id ? Object.assign({}, p, { status: 'Reembolsado' }) : p), confirming: false }));
    this.flash('Reembolso enviado ao emissor');
  }
  copyId() {
    const id = this.state.selectedId;
    if (navigator.clipboard) navigator.clipboard.writeText(id).catch(() => {});
    this.flash('ID copiado para a área de transferência');
  }
  paletteItems() {
    const items = [
      { kind: 'Ir para', label: 'Visão geral', hint: 'G V', run: () => this.go('overview') },
      { kind: 'Ir para', label: 'Pagamentos', hint: 'G P', run: () => this.go('payments') },
      { kind: 'Ir para', label: 'Clientes', hint: 'G C', run: () => this.go('clients') },
      { kind: 'Ação', label: 'Criar nova cobrança', hint: 'C', run: () => this.setState({ chargeOpen: true, paletteOpen: false }) },
      { kind: 'Ação', label: 'Exportar pagamentos em CSV', hint: '', run: () => { this.setState({ paletteOpen: false }); this.exportCsv(); } },
      { kind: 'Filtro', label: 'Só pagamentos recusados', hint: '', run: () => this.setState({ view: 'payments', status: 'Recusado', paletteOpen: false }) },
      { kind: 'Filtro', label: 'Só pagamentos em análise', hint: '', run: () => this.setState({ view: 'payments', status: 'Em análise', paletteOpen: false }) }
    ];
    this.state.payments.slice(0, 14).forEach(p => items.push({
      kind: 'Cliente', label: p.customer + ' · ' + this.brl(p.amount), hint: p.id.slice(0, 9),
      run: () => this.setState({ view: 'payments', selectedId: p.id, paletteOpen: false, confirming: false })
    }));
    const q = this.state.paletteQuery.trim().toLowerCase();
    return q ? items.filter(i => (i.label + ' ' + i.kind).toLowerCase().includes(q)) : items.slice(0, 9);
  }
  onPaletteKey(e) {
    const list = this.paletteItems();
    if (e.key === 'ArrowDown') { e.preventDefault(); this.setState(s => ({ paletteIdx: Math.min(list.length - 1, s.paletteIdx + 1) })); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); this.setState(s => ({ paletteIdx: Math.max(0, s.paletteIdx - 1) })); }
    else if (e.key === 'Enter') { const it = list[this.state.paletteIdx]; if (it) it.run(); }
  }

  renderVals() {
    const S = this.state, mc = this.metricConf(), data = this.series();
    const navBase = 'display:flex;align-items:center;gap:10px;padding:8px 10px;border-radius:8px;font-size:13.5px;font-weight:500;width:100%;white-space:nowrap;';
    const navOn = navBase + 'background:#F0F4F3;color:#0A6558;font-weight:600';
    const navOff = navBase + 'color:#475467';
    const navStyles = {
      overview: S.view === 'overview' ? navOn : navOff,
      payments: S.view === 'payments' ? navOn : navOff,
      clients: S.view === 'clients' ? navOn : navOff,
      balances: navOff, reports: navOff, settings: navOff
    };

    // KPIs
    const kpiDefs = [
      { key: 'gross', label: 'Volume bruto', fmt: (v) => this.brl(v), agg: 'sum', delta: 12.4, sub: 'inclui reembolsos' },
      { key: 'net', label: 'Receita líquida', fmt: (v) => this.brl(v), agg: 'sum', delta: 11.1, sub: 'após taxas' },
      { key: 'count', label: 'Pagamentos aprovados', fmt: (v) => this.int(v), agg: 'sum', delta: 8.7, sub: 'ticket ' + this.brl(this.agg(data.map(d => d.gross), 'sum') / this.agg(data.map(d => d.count), 'sum')) },
      { key: 'decline', label: 'Taxa de recusa', fmt: (v) => this.pct(v), agg: 'avg', delta: -0.6, sub: 'meta abaixo de 5%' }
    ];
    const kpis = kpiDefs.map(k => {
      const vals = data.map(d => d[k.key]);
      const total = this.agg(vals, k.agg);
      const good = k.key === 'decline' ? k.delta < 0 : k.delta > 0;
      const sel = S.metric === k.key;
      const spark = vals.slice(-9);
      const mn = Math.min(...spark), mx = Math.max(...spark) || 1;
      return {
        label: k.label, sub: k.sub,
        value: k.fmt(total * (0.02 + 0.98 * S.tween)),
        delta: (k.delta > 0 ? '+' : '') + k.delta.toLocaleString('pt-BR', { maximumFractionDigits: 1 }) + (k.key === 'decline' ? ' p.p.' : '%'),
        deltaStyle: 'font-size:11.5px;font-weight:600;padding:2px 7px;border-radius:6px;background:' + (good ? '#E6F4F1' : '#FFF1F3') + ';color:' + (good ? '#0A6558' : '#B02A41'),
        spark: spark.map((v, i) => (i * (88 / (spark.length - 1))).toFixed(1) + ',' + (26 - ((v - mn) / (mx - mn || 1)) * 24).toFixed(1)).join(' '),
        sparkColor: sel ? '#0E7C6B' : '#C6CDD6',
        cardStyle: 'background:#fff;border:1px solid ' + (sel ? '#0E7C6B' : '#E4E7EC') + ';border-radius:12px;padding:15px 16px 14px;box-shadow:' + (sel ? '0 0 0 3px rgba(14,124,107,.09)' : '0 1px 2px rgba(16,24,40,.04)') + ';transition:border-color .16s,box-shadow .16s;display:block',
        onClick: () => { this.setState({ metric: k.key, hover: null, tween: 0 }); this.tween(); }
      };
    });

    // chart geometry
    const W = S.chartW, H = 300, padT = 14, padB = 10;
    const vals = data.map(d => d[mc.key]);
    const prev = data.map(d => mc.key === 'count' ? d.prevGross / 172 : mc.key === 'decline' ? d.decline * 1.18 : (mc.key === 'net' ? d.prevGross * 0.957 : d.prevGross));
    const top = Math.max(...vals, ...(S.showPrev ? prev : [0])) * 1.12;
    const y = (v) => padT + (1 - v / top) * (H - padT - padB);
    const x = (i) => data.length < 2 ? 0 : i * (W / (data.length - 1));
    const pts = vals.map((v, i) => [x(i), y(v)]);
    const prevPts = prev.map((v, i) => [x(i), y(v)]);

    const gridLines = [0, 1, 2, 3, 4].map(i => {
      const v = top * (1 - i / 4), t = y(v);
      return {
        label: mc.axis(v),
        style: 'position:absolute;left:0;right:0;top:' + t.toFixed(1) + 'px;height:1px;background:' + (i === 4 ? '#DDE2E8' : '#F1F3F6'),
        labelStyle: 'position:absolute;right:0;top:' + (t - 15).toFixed(1) + 'px;font-size:10.5px;color:#667085;font-family:\'IBM Plex Mono\',monospace;background:#fff;padding:0 3px'
      };
    });
    const every = Math.max(1, Math.ceil(data.length / 7));
    const xLabels = data.filter((_, i) => i % every === 0 || i === data.length - 1).map(d => {
      const i = data.indexOf(d), p = data.length < 2 ? 0 : i / (data.length - 1);
      return { label: d.label, style: 'position:absolute;left:' + (p * 100).toFixed(2) + '%;transform:translateX(' + (p === 0 ? '0' : p === 1 ? '-100%' : '-50%') + ');font-size:10.5px;color:#667085;white-space:nowrap' };
    });

    const hi = S.hover, hasHover = hi !== null && hi >= 0 && hi < data.length && !S.loading;
    const hx = hasHover ? x(hi) : 0, hy = hasHover ? y(vals[hi]) : 0;
    const tipW = 214, tipLeft = Math.max(0, Math.min(W - tipW, hx - tipW / 2));

    const total = this.agg(vals, mc.agg);
    const prevTotal = this.agg(prev, mc.agg);
    const dlt = prevTotal ? ((total - prevTotal) / prevTotal) * 100 : 0;
    const goodDelta = mc.key === 'decline' ? dlt < 0 : dlt > 0;
    const rangeLabel = { '7d': 'últimos 7 dias', '30d': 'últimos 30 dias', '90d': 'últimos 90 dias', '12m': 'últimos 12 meses' }[S.range];

    const segBase = 'padding:5px 12px;border-radius:6px;font-size:12.5px;font-weight:500;white-space:nowrap;flex:0 0 auto;';
    const rangeBtns = [['7d', '7 dias'], ['30d', '30 dias'], ['90d', '90 dias'], ['12m', '12 meses']].map(([k, l]) => ({
      label: l, style: segBase + (S.range === k ? 'background:#fff;color:#0F1419;font-weight:600;box-shadow:0 1px 2px rgba(16,24,40,.1)' : 'color:#667085'),
      onClick: () => this.setRange(k)
    }));
    const mBase = 'padding:5px 11px;border-radius:999px;font-size:12px;font-weight:500;white-space:nowrap;flex:0 0 auto;border:1px solid ';
    const metricBtns = [['gross', 'Volume bruto'], ['net', 'Receita líquida'], ['count', 'Aprovados'], ['decline', 'Taxa de recusa']].map(([k, l]) => ({
      label: l, style: mBase + (S.metric === k ? '#0E7C6B;background:#E6F4F1;color:#0A6558;font-weight:600' : '#E4E7EC;background:#fff;color:#667085'),
      onClick: () => { this.setState({ metric: k, hover: null, tween: 0 }); this.tween(); }
    }));

    // methods + failures
    const grossTotal = this.agg(data.map(d => d.gross), 'sum');
    const mShares = [['Pix', 0.412, '#0E7C6B'], ['Cartão de crédito', 0.371, '#4C8FE0'], ['Boleto', 0.126, '#DE9B26'], ['Cartão de débito', 0.091, '#9B8BD8']];
    const methods = mShares.map(([name, sh, col]) => ({
      name, share: (sh * 100).toFixed(1).replace('.', ',') + '%', value: this.brlShort(grossTotal * sh),
      segStyle: 'flex:' + sh + ';background:' + col + ';border-radius:5px',
      dotStyle: 'width:8px;height:8px;border-radius:3px;background:' + col + ';flex:0 0 auto'
    }));
    const failBase = Math.round(grossTotal / 2600);
    const fShares = [['Saldo insuficiente', 0.38], ['Cartão expirado', 0.23], ['Suspeita de fraude', 0.17], ['Dados incorretos', 0.13], ['Limite excedido', 0.09]];
    const failures = fShares.map(([name, sh]) => ({
      name, count: this.int(failBase * sh), share: (sh * 100).toFixed(0) + '%',
      barStyle: 'height:100%;width:' + (sh * 100 / 0.38 * S.tween).toFixed(1) + '%;background:#D6455F;border-radius:4px;transition:width .5s cubic-bezier(.22,1,.36,1)'
    }));

    const open = (p) => () => this.setState({ selectedId: p.id, confirming: false });
    const mkRow = (p) => ({
      id: p.id, customer: p.customer, email: p.email, amount: this.brl(p.amount), status: p.status,
      method: p.method, date: this.fmtDate(p.date), pillStyle: this.pill(p.status),
      initials: this.initials(p.customer), avatarStyle: this.avatar(p.customer), open: open(p)
    });

    const rowsAll = this.filtered();
    const rows = rowsAll.slice(0, S.limit).map(mkRow);
    const recent = this.state.payments.slice(0, 5).map(mkRow);

    // clients
    const byName = {};
    this.state.payments.forEach(p => {
      if (!byName[p.customer]) byName[p.customer] = { name: p.customer, email: p.email, count: 0, total: 0, last: p.date };
      const c = byName[p.customer];
      c.count++; c.total += p.amount; if (p.date > c.last) c.last = p.date;
    });
    const cq = S.query.trim().toLowerCase();
    const clients = Object.values(byName)
      .filter(c => !cq || (c.name + ' ' + c.email).toLowerCase().includes(cq))
      .sort((a, b) => b.total - a.total)
      .map(c => ({ name: c.name, email: c.email, count: this.int(c.count), total: this.brl(c.total), last: this.fmtDate(c.last), initials: this.initials(c.name), avatarStyle: this.avatar(c.name) }));

    // drawer
    const p = this.state.payments.find(z => z.id === S.selectedId);
    let sel = null;
    if (p) {
      const fee = p.amount * 0.0399 + 0.39;
      const steps = {
        'Aprovado': [['Pagamento criado', -1400], ['Autorizado pelo emissor', -1380], ['Capturado', -1370], ['Liquidação prevista', 172800000]],
        'Recusado': [['Pagamento criado', -1400], ['Recusado pelo emissor', -1360]],
        'Reembolsado': [['Pagamento criado', -260000], ['Capturado', -259000], ['Reembolso solicitado', -4000], ['Reembolso enviado', -3800]],
        'Em análise': [['Pagamento criado', -900], ['Em análise antifraude', -800]]
      }[p.status] || [];
      sel = {
        id: p.id, amount: this.brl(p.amount), status: p.status, pillStyle: this.pill(p.status),
        customer: p.customer, email: p.email, method: p.method, fullDate: this.fullDate(p.date),
        fee: '− ' + this.brl(fee), net: this.brl(p.amount - fee), risk: p.risk,
        timeline: steps.map(([title, off], i) => ({
          title, time: off > 0 ? 'previsto · ' + this.fmtDate(new Date(p.date.getTime() + off)) : this.fmtDate(new Date(p.date.getTime() + off * 1000)),
          dotStyle: 'width:9px;height:9px;border-radius:50%;background:' + (off > 0 ? '#fff' : i === steps.length - 1 ? '#0E7C6B' : '#0E7C6B') + ';border:2px solid ' + (off > 0 ? '#C6CDD6' : '#0E7C6B') + ';flex:0 0 auto;margin-top:4px',
          lineStyle: i === steps.length - 1 ? 'display:none' : 'flex:1;width:2px;background:#E4E7EC;margin:3px 0'
        }))
      };
    }
    const refundable = p && (p.status === 'Aprovado' || p.status === 'Em análise');

    const chipBase = 'padding:6px 12px;border-radius:8px;font-size:12.5px;font-weight:500;white-space:nowrap;flex:0 0 auto;border:1px solid ';
    const statusChips = ['todos', 'Aprovado', 'Em análise', 'Recusado', 'Reembolsado'].map(s => ({
      label: s === 'todos' ? 'Todos' : s,
      style: chipBase + (S.status === s ? '#0F1419;background:#0F1419;color:#fff;font-weight:600' : '#E4E7EC;background:#fff;color:#475467'),
      onClick: () => this.setState({ status: s, limit: this.props.linhasPorPagina || 12 })
    }));

    const palItems = this.paletteItems();
    const paletteResults = palItems.map((it, i) => ({
      kind: it.kind, label: it.label, hint: it.hint,
      style: 'display:flex;align-items:center;gap:10px;width:100%;padding:9px 10px;border-radius:8px;background:' + (i === S.paletteIdx ? '#F4F6F8' : 'transparent'),
      badgeStyle: 'flex:0 0 auto;font-size:10.5px;font-weight:600;letter-spacing:.03em;color:#667085;background:#F1F3F6;border-radius:5px;padding:2px 6px;width:58px;text-align:center',
      onClick: () => it.run(), onHover: () => this.setState({ paletteIdx: i })
    }));

    const optBase = 'padding:7px 13px;border-radius:8px;font-size:12.5px;font-weight:500;white-space:nowrap;flex:0 0 auto;border:1px solid ';
    const methodOptions = ['Pix', 'Cartão de crédito', 'Boleto'].map(m => ({
      label: m, style: optBase + (S.formMethod === m ? '#0E7C6B;background:#E6F4F1;color:#0A6558;font-weight:600' : '#E4E7EC;background:#fff;color:#475467'),
      onClick: () => this.setState({ formMethod: m })
    }));

    return {
      navStyles, totalPayments: this.state.payments.length,
      isOverview: S.view === 'overview', isPayments: S.view === 'payments', isClients: S.view === 'clients',
      viewTitle: { overview: 'Visão geral', payments: 'Pagamentos', clients: 'Clientes' }[S.view],
      viewMeta: S.view === 'overview' ? '· ' + rangeLabel : S.view === 'payments' ? '· ' + this.state.payments.length + ' registros' : '· ' + clients.length + ' cadastrados',
      goOverview: () => this.go('overview'), goPayments: () => this.go('payments'), goClients: () => this.go('clients'),
      soon: () => this.flash('Essa área ainda não existe no protótipo', 'warn'),

      kpis, rangeBtns, metricBtns,
      metricTitle: mc.title + ' · ' + rangeLabel,
      metricValue: mc.fmt(total * (0.02 + 0.98 * S.tween)),
      metricDelta: (dlt > 0 ? '+' : '') + dlt.toFixed(1).replace('.', ',') + '%',
      metricDeltaStyle: 'font-size:12.5px;font-weight:600;padding:3px 8px;border-radius:7px;background:' + (goodDelta ? '#E6F4F1' : '#FFF1F3') + ';color:' + (goodDelta ? '#0A6558' : '#B02A41'),
      metricSub: 'vs ' + mc.fmt(prevTotal) + ' no período anterior',
      togglePrev: () => this.setState(s => ({ showPrev: !s.showPrev })),
      prevBtnStyle: 'display:flex;align-items:center;gap:7px;padding:6px 11px;border-radius:8px;font-size:12.5px;font-weight:500;white-space:nowrap;flex:0 0 auto;border:1px solid ' + (S.showPrev ? '#D0D5DD;background:#fff;color:#0F1419' : '#E4E7EC;background:#fff;color:#667085'),

      loading: S.loading, chartReady: !S.loading,
      chartRef: this.chartRef, chartViewBox: '0 0 ' + W + ' ' + H,
      mainPath: this.path(pts, false, H), areaPath: this.path(pts, true, H - padB),
      prevPath: this.path(prevPts, false, H), prevOpacity: S.showPrev ? 1 : 0,
      gridLines, xLabels,
      onChartMove: (e) => {
        const r = e.currentTarget.getBoundingClientRect();
        const i = Math.round(((e.clientX - r.left) / r.width) * (data.length - 1));
        if (i !== S.hover) this.setState({ hover: Math.max(0, Math.min(data.length - 1, i)) });
      },
      onChartLeave: () => this.setState({ hover: null }),
      hoverActive: hasHover,
      crossStyle: 'position:absolute;top:0;bottom:' + padB + 'px;left:' + hx.toFixed(1) + 'px;width:1px;background:#0E7C6B;opacity:.35;pointer-events:none',
      dotStyle: 'position:absolute;left:' + hx.toFixed(1) + 'px;top:' + hy.toFixed(1) + 'px;width:11px;height:11px;margin:-5.5px 0 0 -5.5px;border-radius:50%;background:#0E7C6B;border:2.5px solid #fff;box-shadow:0 0 0 3px rgba(14,124,107,.16);pointer-events:none',
      tipStyle: 'position:absolute;left:' + tipLeft.toFixed(1) + 'px;top:' + Math.max(4, hy - 118).toFixed(1) + 'px;width:' + tipW + 'px;background:#fff;border:1px solid #E4E7EC;border-radius:10px;box-shadow:0 8px 24px rgba(16,24,40,.13);padding:11px 12px;pointer-events:none',
      tip: hasHover ? {
        label: data[hi].label, value: mc.fmt(vals[hi]), prev: mc.fmt(prev[hi]),
        extra: this.int(data[hi].count) + ' pagamentos · ticket ' + this.brl(data[hi].gross / data[hi].count)
      } : { label: '', value: '', prev: '', extra: '' },

      methods, failures, failTotal: this.int(failBase),
      recent, rows,
      resultLabel: rowsAll.length + ' de ' + this.state.payments.length,
      isEmpty: rowsAll.length === 0, hasMore: rowsAll.length > S.limit,
      loadMore: () => this.setState(s => ({ limit: s.limit + 10 })),
      clearFilters: () => this.setState({ query: '', status: 'todos' }),
      statusChips, query: S.query, setQuery: (e) => this.setState({ query: e.target.value, limit: this.props.linhasPorPagina || 12 }),
      sortCustomer: () => this.sort('customer'), sortAmount: () => this.sort('amount'),
      sortStatus: () => this.sort('status'), sortDate: () => this.sort('date'),
      arrowCustomer: this.arrow('customer'), arrowAmount: this.arrow('amount'),
      arrowStatus: this.arrow('status'), arrowDate: this.arrow('date'),
      exportCsv: () => this.exportCsv(),
      clients, clientCount: clients.length + ' clientes',

      hasSelected: !!sel, sel: sel || { timeline: [] },
      closeDrawer: () => this.setState({ selectedId: null, confirming: false }),
      confirming: S.confirming, showActions: !S.confirming,
      refundStart: () => { if (refundable) this.setState({ confirming: true }); else this.flash('Esse pagamento não pode ser reembolsado', 'warn'); },
      refundConfirm: () => this.refundConfirm(), refundCancel: () => this.setState({ confirming: false }),
      refundLabel: refundable ? 'Reembolsar' : 'Reembolso indisponível',
      refundBtnStyle: 'flex:1;text-align:center;border:1px solid ' + (refundable ? '#F2C8CF;color:#B02A41' : '#E4E7EC;color:#667085') + ';border-radius:8px;padding:9px 0;font-size:13px;font-weight:600;background:#fff',
      copyId: () => this.copyId(),

      chargeOpen: S.chargeOpen, openCharge: () => this.setState({ chargeOpen: true, formError: false }),
      closeCharge: () => this.setState({ chargeOpen: false, formError: false }),
      formAmount: S.formAmount, setFormAmount: (e) => this.setState({ formAmount: e.target.value }),
      formName: S.formName, setFormName: (e) => this.setState({ formName: e.target.value }),
      methodOptions, formError: S.formError, submitting: S.submitting,
      submitLabel: S.submitting ? 'Criando…' : 'Criar cobrança',
      submitCharge: () => this.submitCharge(),

      paletteOpen: S.paletteOpen, paletteRef: this.paletteRef,
      openPalette: () => this.setState({ paletteOpen: true, paletteQuery: '', paletteIdx: 0 }, () => { if (this.paletteRef.current) this.paletteRef.current.focus(); }),
      closePalette: () => this.setState({ paletteOpen: false }),
      paletteQuery: S.paletteQuery,
      setPaletteQuery: (e) => this.setState({ paletteQuery: e.target.value, paletteIdx: 0 }),
      onPaletteKey: (e) => this.onPaletteKey(e),
      paletteResults, paletteEmpty: paletteResults.length === 0,

      hasToast: !!S.toast, toastText: S.toast ? S.toast.text : '',
      toastDotStyle: 'width:7px;height:7px;border-radius:50%;background:' + (S.toast && S.toast.tone === 'warn' ? '#E8A33D' : '#4FD1B5'),
      stop: (e) => e.stopPropagation()
    };
  }
}
